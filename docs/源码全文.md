        /// <summary>改名字时，如果路径框还停在"自动算出来的那个"，就跟着更新。</summary>
        private void SyncSuggestedDir()
        {
            if (_editing != null) { return; }
            if (_rdoDefault.Checked) { return; }
            string current = _txtDir.Text.Trim();
            if (current.Length > 0 && !string.Equals(current, _lastAutoDir, StringComparison.OrdinalIgnoreCase)) { return; }
            string suggested = SuggestedDir();
            _lastAutoDir = suggested;
            if (!string.Equals(current, suggested, StringComparison.OrdinalIgnoreCase)) { _txtDir.Text = suggested; }
        }

        private string SuggestedDir()
        {
            string name = _txtName.Text.Trim();
            if (name.Length == 0) { name = "账号 " + (_settings.Profiles.Count + 1).ToString(); }
            return Path.Combine(_settings.EffectiveProfileRoot, LaunchService.SafeFolderName(name));
        }

        private Profile FindDefault()
        {
            foreach (Profile p in _settings.Profiles) { if (p.IsDefault) { return p; } }
            return null;
        }

        private void Browse()
        {

            {
                dlg.Description = "选择这个账号的数据目录";
                dlg.ShowNewFolderButton = true;
                string current = _txtDir.Text.Trim();
                if (current.Length > 0 && Directory.Exists(current)) { dlg.SelectedPath = current; }
                if (dlg.ShowDialog(this) == DialogResult.OK)
                {
                    _lastAutoDir = "";
                    _txtDir.Text = dlg.SelectedPath;
                }
            }
        }

        private void OnOk()
        {
            string name = _txtName.Text.Trim();
            if (name.Length == 0)
            {
                MessageBox.Show(this, "请填一个名字。", AppPaths.AppName, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }
            foreach (Profile p in _settings.Profiles)
            {
                if (_editing != null && p.Id == _editing.Id) { continue; }
                if (string.Equals(p.Name, name, StringComparison.OrdinalIgnoreCase))
                {
                    MessageBox.Show(this, "已经有一个叫“" + name + "”的账号了，换个名字吧。",
                        AppPaths.AppName, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }
            }

            bool useDefault = _rdoDefault.Checked;
            string dir = useDefault ? AppPaths.BambuDefaultDataDir : _txtDir.Text.Trim();

            if (!useDefault)
            {
                if (dir.Length == 0)
                {
                    MessageBox.Show(this, "请指定数据目录，或者点“默认位置”。", AppPaths.AppName,
                        MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }
                try
                {
                    dir = Path.GetFullPath(dir);
                }
                catch (Exception ex)
                {
                    MessageBox.Show(this, "这个路径不对：" + ex.Message, AppPaths.AppName,
                        MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return;
                }
                foreach (Profile p in _settings.Profiles)
                {
                    if (string.Equals(Path.GetFullPath(p.Dir).TrimEnd('\\'), dir.TrimEnd('\\'), StringComparison.OrdinalIgnoreCase))
                    {
                        MessageBox.Show(this, "这个目录已经被“" + p.Name + "”占用了，换一个吧。",
                            AppPaths.AppName, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                        return;
                    }
                }
            }

            if (_editing == null)
            {
                Profile p = new Profile();
                p.Id = _settings.NextId();
                p.Name = name;
                p.Dir = dir;
                p.IsDefault = useDefault;
                Result = p;

                if (!useDefault)
                {
                    if (_chkCopy.Checked)
                    {
                        string source = AppPaths.BambuDefaultDataDir;
                        if (Directory.Exists(source))
                        {
                            bool existed = Directory.Exists(dir) && Directory.GetFileSystemEntries(dir).Length > 0;
                            if (!existed)
                            {
                                Exception failure = null;
                                using (BusyForm busy = new BusyForm("准备数据目录", "正在复制插件与预设，请稍候…",
                                           delegate { LaunchService.CopySharedData(source, dir, null); }))
                                {
                                    busy.ShowDialog(this);
                                    failure = busy.Error;
                                }
                                if (failure != null)
                                {
                                    Log.Warn("复制插件失败（可忽略）: " + failure.Message);
                                }
                            }
                        }
                    }
                    // 不管有没有复制，目录本身一定得存在，否则启动时会报"NoDir"
                    if (!Directory.Exists(dir))
                    {
                        try { Directory.CreateDirectory(dir); }
                        catch (Exception ex)
                        {
                            MessageBox.Show(this, "创建目录失败：" + ex.Message, AppPaths.AppName,
                                MessageBoxButtons.OK, MessageBoxIcon.Error);
                            return;
                        }
                    }
                }
            }
            else
            {
                _editing.Name = name;
                Result = _editing;
            }

            DialogResult = DialogResult.OK;
            Close();
        }
    }

        /// <summary>把路径变成绝对路径；相对路径按配置目录算（Bambu Studio 会按自己的目录解析相对路径）。</summary>
        internal static string NormalizeDir(string dir)
        {
            if (string.IsNullOrEmpty(dir)) { return ""; }
            string d = dir.Trim();
            if (d.Length == 0) { return ""; }
            try
            {
                if (!Path.IsPathRooted(d)) { d = Path.Combine(AppPaths.ConfigDir, d); }
                return Path.GetFullPath(d);
            }
            catch { return d; }
        }

        /// <summary>比较用的规范化写法（去尾部分隔符，不分大小写比）。</summary>
        private static string DirKey(string dir)
        {
            string d = NormalizeDir(dir);
            return d.TrimEnd('\\');
        }

        internal void IgnoreDir(string dir)
        {
            string key = DirKey(dir);
            if (key.Length == 0) { return; }
            foreach (string old in IgnoredDirs)
            {
                if (string.Equals(DirKey(old), key, StringComparison.OrdinalIgnoreCase)) { return; }
            }
            IgnoredDirs.Add(key);
        }

        internal void UnignoreDir(string dir)
        {
            string key = DirKey(dir);
            if (key.Length == 0) { return; }
            for (int i = IgnoredDirs.Count - 1; i >= 0; i--)
            {
                if (string.Equals(DirKey(IgnoredDirs[i]), key, StringComparison.OrdinalIgnoreCase)) { IgnoredDirs.RemoveAt(i); }
            }
        }

            if (key.Length == 0) { return false; }

    {
        private readonly Dictionary<string, Dictionary<string, string>> _data =

        private readonly List<string> _order = new List<string>();

        internal IEnumerable<string> Sections { get { return _order; } }

        internal static IniFile Load(string path)
        {
            IniFile ini = new IniFile();
            string[] lines = File.ReadAllLines(path, Encoding.UTF8);
            string section = "";
            ini.EnsureSection(section);
            foreach (string raw in lines)
            {
                string line = raw.Trim();
                if (line.Length == 0) { continue; }
                if (line[0] == ';' || line[0] == '#') { continue; }
                if (line[0] == '[')
                {
                    int end = line.IndexOf(']');
                    if (end > 0)
                    {
                        section = line.Substring(1, end - 1).Trim();
                        ini.EnsureSection(section);
                    }
                    continue;
                }
                int eq = line.IndexOf('=');
                if (eq <= 0) { continue; }
                string key = line.Substring(0, eq).Trim();
                string val = line.Substring(eq + 1).Trim();
                ini._data[section][key] = val;
            }
            return ini;
        }

        private void EnsureSection(string section)
        {
            if (!_data.ContainsKey(section))
            {
                _data[section] = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
                _order.Add(section);
            }
        }

        internal void Set(string section, string key, string value)
        {
            EnsureSection(section);
            _data[section][key] = value == null ? "" : value;
        }

        internal void Save(string path)
        {
            StringBuilder sb = new StringBuilder();
            foreach (string section in _order)
            {
                if (section.Length > 0) { sb.AppendLine("[" + section + "]"); }
                foreach (KeyValuePair<string, string> pair in _data[section])
                {
                    sb.AppendLine(pair.Key + "=" + pair.Value);
                }
                if (section.Length > 0) { sb.AppendLine(); }
            }
            File.WriteAllText(path, sb.ToString(), new UTF8Encoding(true));
        }

        internal string Get(string section, string key, string fallback)
        {
            Dictionary<string, string> dict;
            if (_data.TryGetValue(section, out dict))
            {
                string v;
                if (dict.TryGetValue(key, out v)) { return v; }
            }
            return fallback;
        }

        internal int GetInt(string section, string key, int fallback)
        {
            int v;
            if (int.TryParse(Get(section, key, ""), NumberStyles.Integer, CultureInfo.InvariantCulture, out v)) { return v; }
            return fallback;
        }

        internal bool GetBool(string section, string key, bool fallback)
        {
            string v = Get(section, key, "");
            if (v.Length == 0) { return fallback; }
            if (v == "1" || v.Equals("true", StringComparison.OrdinalIgnoreCase) || v.Equals("yes", StringComparison.OrdinalIgnoreCase)) { return true; }
            if (v == "0" || v.Equals("false", StringComparison.OrdinalIgnoreCase) || v.Equals("no", StringComparison.OrdinalIgnoreCase)) { return false; }
            return fallback;
        }

        internal static string PatchSingleInstance(string text, out string problem)
        {
            problem = "";
            if (text == null) { problem = "读不到内容"; return null; }
            int at = text.IndexOf("\"single_instance\"", StringComparison.Ordinal);
            if (at < 0)
            {
                int brace = text.IndexOf('{');
                if (brace < 0) { problem = "内容不是 JSON（找不到左花括号）"; return null; }

                // 看左花括号后面第一个有效字符是不是右花括号：空对象插字段不能带逗号
                int probe = brace + 1;
                while (probe < text.Length && (text[probe] == ' ' || text[probe] == '\t' || text[probe] == '\r' || text[probe] == '\n')) { probe++; }
                bool emptyObject = probe >= text.Length || text[probe] == '}';
                string insert = emptyObject
                    ? "\r\n  \"single_instance\": false\r\n"
                    : "\r\n  \"single_instance\": false,";
                return text.Substring(0, brace + 1) + insert + text.Substring(brace + 1);
            }
            int colon = text.IndexOf(':', at);
            if (colon < 0) { problem = "字段后面没有冒号，格式不完整"; return null; }
            int valueStart = colon + 1;
            while (valueStart < text.Length && (text[valueStart] == ' ' || text[valueStart] == '\t')) { valueStart++; }
            if (valueStart >= text.Length) { problem = "字段后面没有值"; return null; }
            if (string.Compare(text, valueStart, "true", 0, 4, StringComparison.OrdinalIgnoreCase) == 0)
            {
                return text.Substring(0, valueStart) + "false" + text.Substring(valueStart + 4);
            }
            if (string.Compare(text, valueStart, "false", 0, 5, StringComparison.OrdinalIgnoreCase) == 0)
            {
                return text; // 已经是 false：原样返回，调用方会认为"无需改动"
            }
            problem = "single_instance 的值不是 true/false（可能是字符串或数字）";
            return null;
        }

            o.Milliseconds = (int)watch.ElapsedMilliseconds;

            Snapshot after = Snapshot.Take(dir);
            o.DirChanged = !before.SameAs(after);
            if (o.DirChanged) { o.Detail = "目录变化：" + before.Describe() + " -> " + after.Describe(); }
            return o;
        }

        internal sealed class Snapshot
        {
            internal int Files;
            internal bool Conf;
            internal long ConfSize;
            internal bool Logs;

            internal static Snapshot Take(string dir)
            {
                Snapshot s = new Snapshot();
                try
                {
                    if (!Directory.Exists(dir)) { return s; }
                    s.Files = Directory.GetFiles(dir).Length;
                    string conf = Path.Combine(dir, "BambuStudio.conf");
                    s.Conf = File.Exists(conf);
                    if (s.Conf) { s.ConfSize = new FileInfo(conf).Length; }
                    s.Logs = Directory.Exists(Path.Combine(dir, "logs"));
                }
                catch { }
                return s;
            }

            internal bool SameAs(Snapshot other)
            {
                if (other == null) { return false; }
                return Files == other.Files && Conf == other.Conf && ConfSize == other.ConfSize && Logs == other.Logs;
            }

            internal string Describe()
            {
                return "文件 " + Files.ToString() + "，配置 " + (Conf ? "有(" + ConfSize.ToString() + ")" : "无")
                    + "，日志 " + (Logs ? "有" : "无");
            }
        }
    }

        /// <summary>用资源管理器的方式打开一个文件或文件夹，成功时通过 pid 带回进程号。</summary>
        internal static bool ShellOpen(string file, string arguments, string workingDir,
            out int pid, out string error)
        {
            pid = 0;
            error = "";
            try
            {
                NativeMethods.SHELLEXECUTEINFO info = new NativeMethods.SHELLEXECUTEINFO();
                info.cbSize = Marshal.SizeOf(typeof(NativeMethods.SHELLEXECUTEINFO));
                info.fMask = NativeMethods.SEE_MASK_NOCLOSEPROCESS
                           | NativeMethods.SEE_MASK_NOASYNC
                           | NativeMethods.SEE_MASK_FLAG_NO_UI;
                info.lpVerb = "open";
                info.lpFile = file;
                info.lpParameters = string.IsNullOrEmpty(arguments) ? null : arguments;
                info.lpDirectory = string.IsNullOrEmpty(workingDir) ? null : workingDir;
                info.nShow = NativeMethods.SW_SHOWNORMAL;

                if (!NativeMethods.ShellExecuteEx(ref info))
                {
                    int code = Marshal.GetLastWin32Error();
                    error = code == 1223
                        ? "被取消了。"
                        : new Win32Exception(code).Message + "（" + code + "）";
                    return false;
                }

                if (info.hProcess != IntPtr.Zero)
                {
                    pid = (int)NativeMethods.GetProcessId(info.hProcess);
                    NativeMethods.CloseHandle(info.hProcess);
                }
                return true;
            }
            catch (Exception ex)
            {
                error = ex.Message;
                return false;
            }
        }

        internal static bool IsAlive(int pid)
        {
            if (pid <= 0) { return false; }
            try
            {
                using (Process p = Process.GetProcessById(pid))
                {
                    if (p.HasExited) { return false; }
                    return p.ProcessName.IndexOf("bambu", StringComparison.OrdinalIgnoreCase) >= 0;
                }

            catch { return false; }

        /// <summary>拿一个"不是自己启动"的进程的退出码（ShellExecuteEx 启动的进程只能这么取）。</summary>
        internal static bool TryGetExitCode(int pid, out int code)
        {
            code = 0;
            if (pid <= 0) { return false; }
            IntPtr handle = NativeMethods.OpenProcess(NativeMethods.PROCESS_QUERY_LIMITED_INFORMATION, false, pid);
            if (handle == IntPtr.Zero) { return false; }
            try
            {
                uint value;
                if (!NativeMethods.GetExitCodeProcess(handle, out value)) { return false; }
                if (value == 259) { return false; } // STILL_ACTIVE：还在跑
                code = unchecked((int)value);
                return true;
            }
            catch { return false; }
            finally { NativeMethods.CloseHandle(handle); }
        }

        internal static string ExtractDataDir(string commandLine)
        {
            if (string.IsNullOrEmpty(commandLine)) { return ""; }
            int idx = commandLine.IndexOf("--datadir", StringComparison.OrdinalIgnoreCase);
            if (idx < 0) { return ""; }
            // 跳过参数名后面的空白；名字和值之间也允许写等号（--datadir=...）
            int at = idx + 9;
            while (at < commandLine.Length && (commandLine[at] == ' ' || commandLine[at] == '\t')) { at++; }
            if (at < commandLine.Length && commandLine[at] == '=') { at++; }
            while (at < commandLine.Length && (commandLine[at] == ' ' || commandLine[at] == '\t')) { at++; }
            if (at >= commandLine.Length) { return ""; }

            string rest = commandLine.Substring(at);
            if (rest[0] == '"')
            {
                int close = rest.IndexOf('"', 1);
                if (close > 1) { return rest.Substring(1, close - 1); }
                return "";
            }
            int space = rest.IndexOf(' ');
            if (space < 0) { return rest; }
            return rest.Substring(0, space);
        }

                b = Path.GetFullPath(b).TrimEnd('\\');

        internal static LaunchResult Launch(Settings settings, Profile p, LaunchMethod method)
        {
            LaunchResult result = new LaunchResult();
            result.Method = method;

            string exe = settings.StudioExe;
            if (string.IsNullOrEmpty(exe) || !File.Exists(exe))
            {
                exe = StudioLocator.Find(settings.StudioExe);
                if (string.IsNullOrEmpty(exe))
                {
                    result.Status = LaunchStatus.NoExe;
                    result.Message = "没找到 Bambu Studio 主程序，请到“高级设置”里指定 bambu-studio.exe 的位置。";
                    return result;
                }
                settings.StudioExe = exe;
            }

            if (!p.IsDefault && !Directory.Exists(p.Dir))
            {
                result.Status = LaunchStatus.NoDir;
                result.Message = "这个账号的数据目录不见了：\r\n" + p.Dir +
                    "\r\n\r\n可以在“高级设置”里换个目录，或者删掉这个账号重新添加一个。";
                return result;
            }

            if (settings.DedupeLaunch)
            {
                int running = GetRunningPid(p);
                if (running > 0)
                {
                    result.Status = LaunchStatus.AlreadyRunning;
                    result.Pid = running;
                    result.Message = "“" + p.Name + "”已经开着（PID " + running + "），不重复启动。";
                    BringToFront(running, 2);
                    return result;
                }
            }

            string args = "";
            if (!p.IsDefault)
            {
                args = "--datadir \"" + p.Dir + "\"";
            }
            if (!string.IsNullOrEmpty(settings.ExtraArgs))
            {
                args = (args.Length > 0 ? args + " " : "") + settings.ExtraArgs;
            }

            string workDir = "";
            try { workDir = Path.GetDirectoryName(exe); }
            catch { }

            // 老版本（02.07 及更早）的新目录默认 single_instance: true，
            // 新进程会把启动请求转给"不存在的老窗口"然后自己退出，看起来就是"启动不了"。
            // 每次启动前顺手把这一项确认成 false（幂等，改之前会备份）。
            if (!p.IsDefault)
            {
                try
                {
                    string repairNote;
                    if (ProfileRepair.EnsureSingleInstanceOff(p.Dir, out repairNote) && repairNote.Length > 0)
                    {
                        result.RepairNote = repairNote;
                    }
                    if (!HasPlugins(p.Dir))
                    {
                        string extra = "这个账号目录里还没有网络插件，第一次打开时会自动下载，可能要等几十秒。";
                        result.RepairNote = result.RepairNote.Length > 0
                            ? result.RepairNote + " " + extra
                            : extra;
                    }
                }
                catch (Exception ex)
                {
                    Log.Warn("启动前检查 single_instance 失败: " + ex.Message);
                }
            }

            // 首选：和双击快捷方式完全一样的方式（ShellExecuteEx）。
            // 直接 CreateProcess 会在某些版本上让 Bambu Studio 启动几秒后崩掉（0xC0000409），
            // 因为它继承了我们这个 GUI 进程的 std 句柄。
            bool tryShell = method != LaunchMethod.Direct;
            bool tryDirect = method != LaunchMethod.Shell;

            if (tryShell)
            {
                int shellPid;
                string shellError;
                if (NativeMethods.ShellOpen(exe, args, workDir, out shellPid, out shellError) && shellPid > 0)
                {
                    result.Pid = shellPid;
                    result.UsedMethod = LaunchMethod.Shell;
                    WriteRecordedPid(p, shellPid);
                    result.Status = LaunchStatus.Started;
                    result.Message = "已启动“" + p.Name + "”（PID " + shellPid + "）";
                    Log.Info("启动 " + p.Name + " -> " + exe + " " + args + " [" + shellPid + "] 方式=双击");
                    return result;
                }
                if (!string.IsNullOrEmpty(shellError))
                {
                    Log.Warn("ShellExecuteEx 启动失败，改用常规方式：" + shellError);
                }
            }

            if (!tryDirect)
            {
                result.Status = LaunchStatus.Failed;
                result.Message = "没能用“双击方式”启动 Bambu Studio。\r\n\r\n" +
                    "可以在“高级设置 → 启动方式”里换成“常规方式”再试一次。";
                return result;
            }

            try
            {
                ProcessStartInfo psi = new ProcessStartInfo();
                psi.FileName = exe;
                psi.UseShellExecute = false;
                psi.CreateNoWindow = true;
                psi.WorkingDirectory = workDir;
                psi.Arguments = args;

                using (Process proc = Process.Start(psi))
                {
                    if (proc != null)
                    {
                        result.Pid = proc.Id;
                        result.UsedMethod = LaunchMethod.Direct;
                        WriteRecordedPid(p, proc.Id);
                        result.Status = LaunchStatus.Started;
                        result.Message = "已启动“" + p.Name + "”（PID " + proc.Id + "）";
                        Log.Info("启动 " + p.Name + " -> " + exe + " " + args + " 方式=常规");
                    }
                    else
                    {
                        result.Status = LaunchStatus.Failed;
                        result.Message = "启动失败：进程没有创建成功。";
                    }
                }
            }
            catch (Exception ex)
            {
                Log.Error("启动失败: " + p.Name, ex);
                result.Status = LaunchStatus.Failed;
                result.Message = "启动失败：" + ex.Message;
            }
            return result;

        internal static bool HasWindow(int pid, out string title)
        {

            string found = "";
            try
            {
                NativeMethods.EnumWindows(delegate(IntPtr hWnd, IntPtr lParam)
                {
                    uint owner;
                    NativeMethods.GetWindowThreadProcessId(hWnd, out owner);
                    if ((int)owner != pid) { return true; }
                    if (!NativeMethods.IsWindowVisible(hWnd)) { return true; }
                    StringBuilder text = new StringBuilder(512);
                    NativeMethods.GetWindowText(hWnd, text, text.Capacity);
                    if (text.Length > 0)
                    {
                        found = text.ToString();
                        return false;
                    }
                    return true;
                }, IntPtr.Zero);

        /// <summary>
        /// 打开一个文件夹。三条路依次试（shell 打开 / explorer 参数 / cmd start），
        /// 都失败就把原因交回调用方，让界面上能看到，而不是静默什么都不发生。
        /// </summary>
        internal static bool OpenFolder(string path, out string error)
        {
            error = "";
            if (string.IsNullOrEmpty(path))
            {
                error = "路径是空的。";
                return false;
            }
            try
            {
                if (!Directory.Exists(path)) { Directory.CreateDirectory(path); }
            }
            catch (Exception ex)
            {
                error = "目录不存在，而且没法创建：" + ex.Message;
                Log.Error("准备目录失败: " + path, ex);
                return false;
            }

            Exception last = null;
            int pid;
            string shellError;

            // 方式 1：ShellExecuteEx（和双击文件夹一样，最稳，也最少被策略拦）
            if (NativeMethods.ShellOpen(path, "", path, out pid, out shellError))
            {
                ActivateFolderWindow(path, 1500);
                return true;
            }
            if (!string.IsNullOrEmpty(shellError))
            {
                Log.Warn("打开目录（方式1 ShellExecuteEx）失败: " + shellError);
                last = new InvalidOperationException(shellError);
            }

            // 方式 2：explorer.exe 带路径
            try
            {
                ProcessStartInfo psi = new ProcessStartInfo("explorer.exe", "\"" + path + "\"");
                psi.UseShellExecute = true;
                Process.Start(psi);
                ActivateFolderWindow(path, 1500);
                return true;
            }
            catch (Exception ex)
            {
                last = ex;
                Log.Warn("打开目录（方式2）失败: " + ex.Message);
            }

            // 方式 3：cmd 的 start
            try
            {
                ProcessStartInfo psi = new ProcessStartInfo("cmd.exe", "/c start \"\" \"" + path + "\"");
                psi.UseShellExecute = false;
                psi.CreateNoWindow = true;
                Process.Start(psi);
                ActivateFolderWindow(path, 1500);
                return true;
            }
            catch (Exception ex)
            {
                last = ex;
                Log.Warn("打开目录（方式3）失败: " + ex.Message);
            }

            error = last == null ? "系统没有响应打开请求。" : last.Message;
            Log.Error("打开目录全部失败: " + path, last);
            return false;
        }

        /// <summary>
        /// 把刚打开的资源管理器窗口换到前台。
        /// 我们的窗口是模态对话框，explorer 新窗口经常生成在它后面，用户会以为"点了没反应"。
        /// </summary>
        internal static bool ActivateFolderWindow(string folder, int waitMilliseconds)
        {
            string leaf = "";
            try { leaf = Path.GetFileName(folder.TrimEnd('\\', '/')); }
            catch { }
            if (leaf.Length == 0) { leaf = folder; }

            int waited = 0;
            while (waited < waitMilliseconds)
            {
                IntPtr hit = FindFolderWindow(leaf);
                if (hit != IntPtr.Zero)
                {
                    NativeMethods.ShowWindow(hit, NativeMethods.SW_RESTORE);
                    NativeMethods.SetForegroundWindow(hit);
                    return true;
                }
                Thread.Sleep(100);
                waited += 100;
            }
            return false;
        }

        private static IntPtr FindFolderWindow(string leaf)
        {
            IntPtr found = IntPtr.Zero;
            try
            {
                NativeMethods.EnumWindows(delegate(IntPtr hWnd, IntPtr lParam)
                {
                    try
                    {
                        if (!NativeMethods.IsWindowVisible(hWnd)) { return true; }
                        StringBuilder cls = new StringBuilder(64);
                        NativeMethods.GetClassName(hWnd, cls, cls.Capacity);
                        if (!string.Equals(cls.ToString(), "CabinetWClass", StringComparison.OrdinalIgnoreCase))
                        {
                            return true;
                        }
                        StringBuilder title = new StringBuilder(512);
                        NativeMethods.GetWindowText(hWnd, title, title.Capacity);
                        if (title.ToString().IndexOf(leaf, StringComparison.OrdinalIgnoreCase) >= 0)
                        {
                            found = hWnd;
                            return false;
                        }
                    }
                    catch { }
                    return true;
                }, IntPtr.Zero);
            }
            catch { }
            return found;
        }

        /// <summary>安全的名字 -> 目录名。</summary>
        internal static string SafeFolderName(string name)
        {
            if (string.IsNullOrEmpty(name)) { return "profile"; }
            System.Text.StringBuilder sb = new System.Text.StringBuilder();
            char[] invalid = Path.GetInvalidFileNameChars();
            foreach (char c in name)
            {
                bool bad = false;
                foreach (char ic in invalid) { if (c == ic) { bad = true; break; } }
                if (bad || c == ' ') { sb.Append('_'); } else { sb.Append(c); }
            }
            // 只去掉首尾的空格和点：尾部的下划线是用户名字的一部分，别削掉
            string s = sb.ToString().Trim(' ', '.');
            if (s.Length == 0) { s = "profile"; }
            if (IsReservedName(s)) { s = "_" + s; }
            if (s.Length > 40) { s = s.Substring(0, 40); }
            return s;
        }

        /// <summary>Windows 保留设备名（CON、NUL、COM1…）不能当文件/目录名。</summary>
        private static bool IsReservedName(string name)
        {
            string bare = name;
            int dot = bare.IndexOf('.');
            if (dot >= 0) { bare = bare.Substring(0, dot); }
            bare = bare.Trim().ToUpperInvariant();
            if (bare.Length == 0) { return false; }
            if (bare == "CON" || bare == "PRN" || bare == "AUX" || bare == "NUL") { return true; }
            if (bare.Length == 4 &&
                (bare.StartsWith("COM", StringComparison.Ordinal) || bare.StartsWith("LPT", StringComparison.Ordinal)))
            {
                char digit = bare[3];
                if (digit >= '1' && digit <= '9') { return true; }
            }
            return false;
        }

        {
            List<Found> result = new List<Found>();

            HashSet<string> rootSeen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            HashSet<string> dirSeen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

                {

                        if (string.Equals(name, "profiles", StringComparison.OrdinalIgnoreCase))

                            addRoot(sub);
                        }
                    }
                }
                catch { }

                    Found f = new Found();

                    f.Dir = full;

        /// <summary>把找到的目录并进配置（已存在的跳过）。返回新加入的个数。</summary>
        internal static int Import(Settings settings)
        {
            int added = 0;
            List<Found> found = Scan(settings);
            foreach (Found f in found)
            {
                // 用户明确删掉过的目录，不要再"认"回来
                if (settings.IsIgnored(f.Dir)) { continue; }
                bool known = false;
                foreach (Profile p in settings.Profiles)
                {
                    if (SameDir(p.Dir, f.Dir)) { known = true; break; }
                }
                if (known) { continue; }
                if (SameDir(f.Dir, AppPaths.BambuDefaultDataDir)) { continue; }

                Profile np = new Profile();
                np.Id = settings.NextId();
                np.Name = UniqueName(settings, FancyName(f.Name));
                np.Dir = f.Dir;
                np.IsDefault = false;
                np.Note = "从以前的安装里找到";
                settings.Profiles.Add(np);
                added++;
                Log.Info("导入历史账号目录: " + f.Dir);
            }
            if (added > 0) { settings.Save(); }
            return added;

            base.OnShown(e);

        private void BeginFirstRun()
        {
            using (FirstRunForm form = new FirstRunForm(_settings))
            {
                if (form.ShowDialog(this) != DialogResult.OK)
                {
                    Close();
                    return;
                }
                _settings = form.Settings;
                _settings.FirstRunDone = true;
                SaveSettings();
                RefreshList();
                if (form.LaunchNow && _settings.Profiles.Count > 0)
                {
                    // 只开默认账号：一下子把好几个窗口全弹出来太吓人
                    Profile first = null;
                    foreach (Profile p in _settings.Profiles)
                    {
                        if (p.IsDefault) { first = p; break; }
                    }
                    if (first == null) { first = _settings.Profiles[0]; }
                    SelectById(first.Id);
                    LaunchSelected();
                }
                else
                {
                    SetHint("设置完成。以后双击账号那一行就能直接打开。", Ui.Ok);
                }
            }
        }

        private void RefreshList()
        {
            Profile current = CurrentProfile();
            int keepId = current != null ? current.Id : _settings.LastProfileId;

            // 整个过程只查一次进程表，然后原地更新行，不再整表清空重画（那是闪烁的来源）
            List<RunningInstance> instances = LaunchService.QueryInstances();

            _list.BeginUpdate();
            try
            {
                for (int i = _list.Items.Count - 1; i >= 0; i--)
                {
                    Profile old = _list.Items[i].Tag as Profile;
                    if (old == null || _settings.FindById(old.Id) == null) { _list.Items.RemoveAt(i); }
                }

                int index = 0;
                foreach (Profile p in _settings.Profiles)
                {
                    int pid = LaunchService.GetRunningPid(p, instances);
                    string runText = pid > 0 ? "运行中" : "未运行";
                    string loginText = p.HasLogin ? "已登录" : "未登录";
                    Color runColor = pid > 0 ? Ui.Ok : Ui.TextDim;
                    Color loginColor = p.HasLogin ? Ui.Text : Ui.Warn;

                    ListViewItem item = FindItem(p.Id);
                    if (item == null)
                    {
                        item = new ListViewItem(p.Name);
                        item.SubItems.Add(runText);
                        item.SubItems.Add(loginText);
                        item.SubItems.Add(p.ShortDir);
                        item.Tag = p;
                        item.UseItemStyleForSubItems = false;
                        item.SubItems[1].ForeColor = runColor;
                        item.SubItems[2].ForeColor = loginColor;
                        _list.Items.Insert(Math.Min(index, _list.Items.Count), item);
                    }
                    else
                    {
                        UpdateCell(item, 0, p.Name, null);
                        UpdateCell(item, 1, runText, runColor);
                        UpdateCell(item, 2, loginText, loginColor);
                        UpdateCell(item, 3, p.ShortDir, null);
                        if (item.Index != index)
                        {
                            _list.Items.Remove(item);
                            _list.Items.Insert(Math.Min(index, _list.Items.Count), item);
                        }
                    }
                    index++;
                }

                if (_list.SelectedItems.Count == 0 && keepId > 0)
                {
                    ListViewItem keep = FindItem(keepId);
                    if (keep != null) { keep.Selected = true; }
                }
            }
            finally
            {
                _list.EndUpdate();
            }

            ResizeColumnsIfNeeded();
            UpdateStudioLabel();
            UpdateButtons();
            if (_settings.Profiles.Count == 0)
            {
                SetHint("还没有账号。点上面的“添加账号”，或者关掉程序重新打开走一遍向导。", Ui.Warn);
            }
        }

        private ListViewItem FindItem(int profileId)
        {
            foreach (ListViewItem item in _list.Items)
            {
                Profile p = item.Tag as Profile;
                if (p != null && p.Id == profileId) { return item; }
            }
            return null;
        }

        /// <summary>只在文字真的变了的时候才改，避免每 2.5 秒无谓重绘。</summary>
        private static void UpdateCell(ListViewItem item, int sub, string text, Color? color)
        {
            if (sub >= item.SubItems.Count) { return; }
            ListViewItem.ListViewSubItem cell = item.SubItems[sub];
            if (cell.Text != text) { cell.Text = text; }
            if (color.HasValue && cell.ForeColor != color.Value) { cell.ForeColor = color.Value; }
        }

        private void UpdateStudioLabel()
        {
            string exe = _settings.StudioExe;
            if (string.IsNullOrEmpty(exe) || !File.Exists(exe))
            {
                exe = StudioLocator.Find(_settings.StudioExe);
            }
            if (string.IsNullOrEmpty(exe))
            {
                _lblStudio.Text = "没有找到 Bambu Studio（点“高级设置”可以手动指定）";
                _lblStudio.ForeColor = Ui.Warn;
                return;
            }
            string version = StudioLocator.VersionOf(exe);
            _lblStudio.Text = "Bambu Studio：" + exe + (version.Length > 0 ? "  版本 " + version : "");
            _lblStudio.ForeColor = Ui.TextDim;
        }

        // ------------------------------------------------------------------ 动作

        private void LaunchSelected()
        {
            Profile p = CurrentProfile();
            if (p == null) { return; }
            _settings.LastProfileId = p.Id;
            LaunchResult r = LaunchService.Launch(_settings, p);
            SaveSettings();

            if (r.Status == LaunchStatus.Started)
            {
                if (r.RepairNote.Length > 0)
                {
                    SetHint("“" + p.Name + "”" + r.RepairNote + " 正在唤起窗口…", Ui.Ok);
                }
                else
                {
                    SetHint(r.Message + "，正在唤起窗口…", Ui.Ok);
                }
                ScheduleVerify(p, r.Pid, r.UsedMethod);
                ScheduleFocus(r.Pid);
            }
            else if (r.Status == LaunchStatus.AlreadyRunning)
            {
                SetHint(r.Message, Ui.TextDim);
                ScheduleFocus(r.Pid);
            }
            else
            {
                SetHint(r.Message, Ui.Danger);
                MessageBox.Show(this, r.Message, AppPaths.AppName, MessageBoxButtons.OK, MessageBoxIcon.Warning);
            }
            RefreshList();
        }

        private void LaunchAll()
        {
            if (_settings.Profiles.Count == 0) { return; }
            int started = 0;
            List<string> problems = new List<string>();
            foreach (Profile p in _settings.Profiles)
            {
                LaunchResult r = LaunchService.Launch(_settings, p);
                if (r.Status == LaunchStatus.Started) { started++; ScheduleVerify(p, r.Pid, r.UsedMethod); }
                else if (r.Status == LaunchStatus.AlreadyRunning) { }
                else { problems.Add(p.Name + "：" + r.Message); }
            }
            SaveSettings();
            if (problems.Count > 0)
            {
                SetHint("启动了 " + started + " 个账号，" + problems.Count + " 个有问题。", Ui.Warn);
                MessageBox.Show(this, string.Join("\r\n\r\n", problems.ToArray()), "有账号没有启动",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
            }
            else
            {
                SetHint("已启动 " + started + " 个账号窗口。", Ui.Ok);
            }
            RefreshList();
        }

        private void AddAccount()
        {
            using (AccountForm form = new AccountForm(_settings, null))
            {
                if (form.ShowDialog(this) != DialogResult.OK) { return; }
                Profile created = form.Result;
                if (created == null) { return; }
                _settings.Profiles.Add(created);
                _settings.UnignoreDir(created.Dir);
                SortProfiles();
                _settings.LastProfileId = created.Id;
                SaveSettings();
                RefreshList();

                string ask = "“" + created.Name + "”准备好了。\r\n\r\n现在打开它的窗口吗？" +
                             "打开后请用这个账号对应的拓竹账号登录一次，以后就不用再登录了。";
                if (MessageBox.Show(this, ask, AppPaths.AppName, MessageBoxButtons.YesNo,
                        MessageBoxIcon.Question) == DialogResult.Yes)
                {
                    SelectById(created.Id);
                    LaunchSelected();
                }
            }
        }

        private void SortProfiles()
        {
            _settings.Profiles.Sort(delegate(Profile a, Profile b)
            {
                if (a.IsDefault != b.IsDefault) { return a.IsDefault ? -1 : 1; }
                return a.Id.CompareTo(b.Id);
            });
        }

        private void SelectById(int id)
        {
            foreach (ListViewItem item in _list.Items)
            {
                Profile p = item.Tag as Profile;
                if (p != null && p.Id == id)
                {
                    _list.SelectedItems.Clear();
                    item.Selected = true;
                    item.Focused = true;
                    return;
                }
            }
        }

        private void RenameAccount()
        {
            Profile p = CurrentProfile();
            if (p == null) { return; }
            string name = PromptForm.Ask(this, "重命名账号", "新的名字：", p.Name);
            if (name == null) { return; }
            name = name.Trim();
            if (name.Length == 0) { return; }
            foreach (Profile other in _settings.Profiles)
            {
                if (other == p) { continue; }
                if (string.Equals(other.Name, name, StringComparison.OrdinalIgnoreCase))
                {
                    MessageBox.Show(this, "已经有一个叫“" + name + "”的账号了，换个名字吧。",
                        AppPaths.AppName, MessageBoxButtons.OK, MessageBoxIcon.Information);
                    return;
                }
            }
            p.Name = name;
            SaveSettings();
            RefreshList();
        }

        private void RemoveAccount()
        {
            Profile p = CurrentProfile();
            if (p == null) { return; }
            if (p.IsDefault)
            {
                MessageBox.Show(this, "这是“默认账号”，不能删除。\r\n它对应 Bambu Studio 自己的数据目录，删掉会影响你原来的使用。",
                    AppPaths.AppName, MessageBoxButtons.OK, MessageBoxIcon.Information);
                return;
            }
            string text = "确定要移除“" + p.Name + "”吗？\r\n\r\n数据目录：" + p.Dir;
            if (MessageBox.Show(this, text, AppPaths.AppName, MessageBoxButtons.YesNo,
                    MessageBoxIcon.Question) != DialogResult.Yes)
            {
                return;
            }

            bool insideRoot = false;
            try
            {
                string root = Path.GetFullPath(_settings.EffectiveProfileRoot).TrimEnd('\\') + "\\";
                insideRoot = Path.GetFullPath(p.Dir).StartsWith(root, StringComparison.OrdinalIgnoreCase);
            }
            catch { }

            if (insideRoot)
            {
                DialogResult del = MessageBox.Show(this,
                    "顺便把这个账号的数据目录也删掉吗？\r\n\r\n删掉以后它记住的登录状态和本地设置都会消失，需要重新登录。",
                    AppPaths.AppName, MessageBoxButtons.YesNoCancel, MessageBoxIcon.Warning);
                if (del == DialogResult.Cancel) { return; }
                if (del == DialogResult.Yes)
                {
                    try { Directory.Delete(p.Dir, true); }
                    catch (Exception ex)
                    {
                        MessageBox.Show(this, "删除数据目录失败：" + ex.Message, AppPaths.AppName,
                            MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    }
                }
            }
            _settings.Profiles.Remove(p);
            // 记下来：这个目录是用户主动删掉的，以后"认旧数据"不许再捡回来
            _settings.IgnoreDir(p.Dir);
            SaveSettings();
            RefreshList();
        }

        private void OpenFolderSafely(string path)

            SetHint("打不开这个目录：" + error, Ui.Danger);
            MessageBox.Show(this,
                "打不开这个目录：\r\n" + path + "\r\n\r\n" + error +
                "\r\n\r\n可以把下面这行复制到资源管理器的地址栏里：\r\n" + path,
                AppPaths.AppName, MessageBoxButtons.OK, MessageBoxIcon.Warning);
        }

        /// <summary>Bambu Studio 出窗口可能要十几秒；轮询放到计时器里，别把界面卡住。</summary>
        private void ScheduleFocus(int pid)
        {
            if (pid <= 0) { return; }
            if (LaunchService.TryFocusOnce(pid)) { return; }
            _focusPid = pid;
            _focusDeadline = DateTime.Now.AddSeconds(25);
            if (_focusTimer == null)
            {
                _focusTimer = new Timer();
                _focusTimer.Interval = 400;
                _focusTimer.Tick += delegate { OnFocusTick(); };
            }
            _focusTimer.Start();
        }

        private void OnFocusTick()
        {
            if (_focusPid <= 0 || DateTime.Now > _focusDeadline || !LaunchService.IsAlive(_focusPid))
            {
                StopFocusTimer();
                return;
            }
            if (LaunchService.TryFocusOnce(_focusPid)) { StopFocusTimer(); }
        }

        private void StopFocusTimer()
        {
            _focusPid = 0;
            if (_focusTimer != null) { _focusTimer.Stop(); }
        }

        private void ScheduleVerify(Profile p, int pid, LaunchMethod used)
        {
            if (pid <= 0) { return; }
            StartedOnce item = new StartedOnce();
            item.Profile = p;
            item.Pid = pid;
            item.Used = used;
            _starting.Add(item);
            if (_verifyTimer == null)
            {
                _verifyTimer = new Timer();
                _verifyTimer.Interval = 7000;
                _verifyTimer.Tick += delegate { OnVerifyTick(); };
            }
            _verifyTimer.Stop();
            _verifyTimer.Start();
        }

        private void OnVerifyTick()
        {
            _verifyTimer.Stop();
            List<StartedOnce> dead = new List<StartedOnce>();
            foreach (StartedOnce item in _starting)
            {
                if (!LaunchService.IsAlive(item.Pid)) { dead.Add(item); }
            }
            _starting.Clear();
            if (dead.Count == 0) { return; }

            // 先自动换一种启动方式再试一次：不同版本的 Bambu Studio 认的方式不一样。
            List<StartedOnce> stillDead = new List<StartedOnce>();
            foreach (StartedOnce item in dead)
            {
                if (item.Retried)
                {
                    stillDead.Add(item);
                    continue;
                }
                LaunchMethod other = item.Used == LaunchMethod.Shell ? LaunchMethod.Direct : LaunchMethod.Shell;
                LaunchResult again = LaunchService.Launch(_settings, item.Profile, other);
                if (again.Status == LaunchStatus.Started)
                {
                    SetHint("“" + item.Profile.Name + "”第一次没起来，换了一种启动方式，正在重新尝试…", Ui.Warn);
                    ScheduleVerify(item.Profile, again.Pid, again.UsedMethod);
                }
                else
                {
                    item.Retried = true;
                    stillDead.Add(item);
                }
            }
            if (stillDead.Count == 0) { return; }

            Profile first = stillDead[0].Profile;
            SetHint("“" + first.Name + "”的窗口没有起来。可以右键这一行，点“诊断启动问题”。", Ui.Danger);
            DialogResult answer = MessageBox.Show(this,
                stillDead.Count == 1

using System.Diagnostics;

namespace BambuMultiLauncher
{
    /// <summary>
    /// 负责找到 Bambu Studio 主程序。
    /// 老版本/装在奇怪位置的情况不少（比如按用户安装、装在 D 盘、卸载表里名字不含 Bambu），
    /// 所以这里除了常见目录，还会把注册表翻一遍。
    /// </summary>
    internal static class StudioLocator
    {
        internal const string ExeName = "bambu-studio.exe";

        internal static readonly string[] RelativePaths = new string[]
        {
            @"Bambu Studio\bambu-studio.exe",
            @"BambuStudio\bambu-studio.exe",
            @"Programs\Bambu Studio\bambu-studio.exe",
            @"Bambu Studio\BambuStudio.exe",
            @"BambuLab\Bambu Studio\bambu-studio.exe"

        /// <summary>按优先级返回所有可能的位置（只含真实存在的文件）。</summary>
        internal static List<string> FindAll(string preferred)
        {
            List<string> found = new List<string>();
            HashSet<string> seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

            Action<string> push = delegate(string path)
            {
                if (string.IsNullOrEmpty(path)) { return; }
                try
                {
                    string full = Path.GetFullPath(path.Trim().Trim('"', ' '));
                    if (!File.Exists(full)) { return; }
                    if (!seen.Add(full)) { return; }
                    found.Add(full);
                }
                catch { }
            };

            push(preferred);

            foreach (string fromRegistry in RegistryCandidates())
            {
                push(fromRegistry);
            }

            foreach (string dir in SearchRoots())
            {
                foreach (string rel in RelativePaths) { push(Path.Combine(dir, rel)); }
                push(Path.Combine(dir, "Bambu Studio", ExeName));
            }

            foreach (string install in ReadUninstallLocations())
            {
                push(Path.Combine(install, ExeName));
                foreach (string rel in RelativePaths) { push(Path.Combine(install, rel)); }
            }

            return Order(found, preferred);
        }

        internal static string Find(string preferred)
        {
            List<string> all = FindAll(preferred);
            if (all.Count > 0) { return all[0]; }
            return "";
        }

        private static readonly string[] KnownValueKeys = new string[]
        {
            @"SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\bambu-studio.exe",
            @"SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\App Paths\bambu-studio.exe",
            @"SOFTWARE\Classes\Applications\bambu-studio.exe\shell\open\command",
            @"SOFTWARE\WOW6432Node\Classes\Applications\bambu-studio.exe\shell\open\command"
        };

        private static List<string> RegistryCandidates()
        {
            List<string> list = new List<string>();
            RegistryKey[] hives = new RegistryKey[] { Registry.LocalMachine, Registry.CurrentUser };

            // 1) App Paths / Classes\Applications 这类明确登记过的地方
            foreach (RegistryKey hive in hives)
            {
                foreach (string key in KnownValueKeys)
                {
                    try
                    {
                        using (RegistryKey k = hive.OpenSubKey(key))
                        {
                            if (k == null) { continue; }
                            string def = Convert.ToString(k.GetValue(""));
                            AddPathsFromText(list, def);
                            string dir = Convert.ToString(k.GetValue("Path"));
                            if (!string.IsNullOrEmpty(dir)) { AddPathsFromText(list, dir); }
                        }
                    }
                    catch { }
                }
            }

            // 2) 卸载表（HKLM/HKCU × 64/32 四份）：DisplayName 含 Bambu，或者 DisplayIcon 指向它
            string[] uninstallRoots = new string[]
            {
                @"SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall",
                @"SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall"
            };
            foreach (RegistryKey hive in hives)
            {
                foreach (string root in uninstallRoots)
                {
                    try
                    {
                        using (RegistryKey k = hive.OpenSubKey(root))
                        {
                            if (k == null) { continue; }
                            foreach (string sub in k.GetSubKeyNames())
                            {
                                try
                                {
                                    using (RegistryKey sk = k.OpenSubKey(sub))
                                    {
                                        if (sk == null) { continue; }
                                        string name = Convert.ToString(sk.GetValue("DisplayName"));
                                        string icon = Convert.ToString(sk.GetValue("DisplayIcon"));
                                        string loc = Convert.ToString(sk.GetValue("InstallLocation"));
                                        string uninstall = Convert.ToString(sk.GetValue("UninstallString"));
                                        bool relevant = (name != null && name.IndexOf("bambu", StringComparison.OrdinalIgnoreCase) >= 0)
                                            || (icon != null && icon.IndexOf(ExeName, StringComparison.OrdinalIgnoreCase) >= 0);
                                        if (!relevant) { continue; }
                                        AddPathsFromText(list, loc);
                                        AddPathsFromText(list, icon);
                                        AddPathsFromText(list, uninstall);
                                    }
                                }
                                catch { }
                            }
                        }
                    }
                    catch { }
                }
            }

            // 3) 厂商键：SOFTWARE（含 WOW6432Node）下名字带 bambu 的子树，往下三层把字符串值都看一遍
            foreach (RegistryKey hive in hives)
            {
                SweepVendor(hive, @"SOFTWARE", 0, list);
                SweepVendor(hive, @"SOFTWARE\WOW6432Node", 0, list);
            }
            return list;

        private static void SweepVendor(RegistryKey hive, string path, int depth, List<string> list)
        {
            if (depth > 3) { return; }
            RegistryKey key = null;
            try
            {
                key = hive.OpenSubKey(path);
                if (key == null) { return; }

                if (depth > 0)
                {
                    foreach (string valueName in key.GetValueNames())
                    {
                        try { AddPathsFromText(list, Convert.ToString(key.GetValue(valueName))); }
                        catch { }
                    }
                }

                foreach (string sub in key.GetSubKeyNames())
                {
                    bool looksRelated = sub.IndexOf("bambu", StringComparison.OrdinalIgnoreCase) >= 0;
                    if (!looksRelated && depth >= 1) { continue; }
                    SweepVendor(hive, path + "\\" + sub, depth + 1, list);
                }
            }
            catch { }
            finally
            {
                if (key != null) { key.Close(); }
            }
        }

        /// <summary>从一段注册表文本里抠出可用的 exe 路径或安装目录。</summary>
        private static void AddPathsFromText(List<string> list, string text)
        {
            if (string.IsNullOrEmpty(text)) { return; }

            // 直接就是目录
            try
            {
                string t = text.Trim().Trim('"');
                if (Directory.Exists(t)) { list.Add(Path.Combine(t, ExeName)); }
            }
            catch { }

            // 文本里出现了 bambu-studio.exe：往前找到引号或空格，截出完整路径
            int from = 0;
            while (true)
            {
                int at = text.IndexOf(ExeName, from, StringComparison.OrdinalIgnoreCase);
                if (at < 0) { break; }
                int end = at + ExeName.Length;
                int start = 0;
                for (int i = at - 1; i >= 0; i--)
                {
                    char c = text[i];
                    if (c == '"' || c == '=' || c == ';' || c == '|')
                    {
                        start = i + 1;
                        break;
                    }
                }
                string candidate = text.Substring(start, end - start).Trim().Trim('"');
                if (candidate.Length > 3 && File.Exists(candidate)) { list.Add(candidate); }
                from = end;
            }

            // DisplayIcon 常见形式 "C:\...\bambu-studio.exe,0"
            int comma = text.IndexOf(',');
            if (comma > 0 && text.IndexOf(ExeName, StringComparison.OrdinalIgnoreCase) > 0)
            {
                string head = text.Substring(0, comma).Trim().Trim('"');
                try { if (File.Exists(head)) { list.Add(head); } }
                catch { }
            }
        }

        private static IEnumerable<string> SearchRoots()
        {
            List<string> roots = new List<string>();
            HashSet<string> seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

            Action<string> add = delegate(string p)
            {
                if (string.IsNullOrEmpty(p)) { return; }
                try
                {
                    string full = Path.GetFullPath(p).TrimEnd('\\');
                    if (seen.Add(full) && Directory.Exists(full)) { roots.Add(full); }
                }
                catch { }
            };
            Action<Environment.SpecialFolder> addSpecial = delegate(Environment.SpecialFolder f)
            {
                try { add(Environment.GetFolderPath(f)); }
                catch { }
            };

            addSpecial(Environment.SpecialFolder.ProgramFiles);
            addSpecial(Environment.SpecialFolder.ProgramFilesX86);
            addSpecial(Environment.SpecialFolder.LocalApplicationData);
            addSpecial(Environment.SpecialFolder.ApplicationData);

            // 别的盘上也可能装了：只试几个固定的名字，不做全盘遍历
            try
            {
                foreach (DriveInfo drive in DriveInfo.GetDrives())
                {
                    if (drive.DriveType != DriveType.Fixed || !drive.IsReady) { continue; }
                    string root = drive.RootDirectory.FullName;
                    add(root);
                    add(Path.Combine(root, "Program Files"));
                    add(Path.Combine(root, "Program Files (x86)"));
                    add(Path.Combine(root, "Bambu Studio"));
                    add(Path.Combine(root, "BambuLab"));
                    add(Path.Combine(root, "software"));
                }
            }
            catch { }
            return roots;
        }

        // ------------------------------------------------------------ 杂项

        private static List<string> Order(List<string> found, string preferred)
        {
            if (found.Count <= 1) { return found; }
            List<string> ordered = new List<string>(found);
            ordered.Sort(delegate(string a, string b)
            {
                bool pa = SamePath(a, preferred);
                bool pb = SamePath(b, preferred);
                if (pa != pb) { return pa ? -1 : 1; }
                int v = CompareVersion(VersionOf(b), VersionOf(a));
                if (v != 0) { return v; }
                return string.Compare(a, b, StringComparison.OrdinalIgnoreCase);
            });
            return ordered;
        }

        private static bool SamePath(string a, string b)
        {
            if (string.IsNullOrEmpty(a) || string.IsNullOrEmpty(b)) { return false; }
            try
            {
                return string.Equals(Path.GetFullPath(a), Path.GetFullPath(b), StringComparison.OrdinalIgnoreCase);
            }
            catch { return false; }
        }

        private static int CompareVersion(string a, string b)
        {
            int[] x = Split(a);
            int[] y = Split(b);
            int n = Math.Max(x.Length, y.Length);
            for (int i = 0; i < n; i++)
            {
                int xi = i < x.Length ? x[i] : 0;
                int yi = i < y.Length ? y[i] : 0;
                if (xi != yi) { return xi < yi ? -1 : 1; }
            }
            return 0;
        }

        private static int[] Split(string version)
        {
            List<int> parts = new List<int>();
            if (!string.IsNullOrEmpty(version))
            {
                foreach (string piece in version.Split('.'))
                {
                    int value;
                    if (int.TryParse(piece.Trim(), out value)) { parts.Add(value); }
                    else { parts.Add(0); }
                }
            }
            return parts.ToArray();
        }

        internal static string VersionOf(string exe)
        {
            try
            {
                if (string.IsNullOrEmpty(exe) || !File.Exists(exe)) { return ""; }
                FileVersionInfo info = FileVersionInfo.GetVersionInfo(exe);
                if (!string.IsNullOrEmpty(info.ProductVersion)) { return info.ProductVersion; }
                return info.FileVersion;
            }
            catch { return ""; }
        }

        /// <summary>候选目录里是否已有 Bambu Studio 的数据（判断“这台机器装没用过”）。</summary>
        internal static bool HasExistingData()
        {
            try
            {
                string dir = AppPaths.BambuDefaultDataDir;
                return Directory.Exists(dir) && File.Exists(Path.Combine(dir, "BambuStudio.conf"));
            }
            catch { return false; }
        }
    }
